def test_add_three_items_and_remove_one(cart_with):
    cart_page = cart_with(
        "sauce-labs-backpack",
        "sauce-labs-bike-light",
        "sauce-labs-bolt-t-shirt",
    )

    expect(cart_page.cart_items).to_have_count(3)
    expect(cart_page.item_names).to_have_text(
        [
            "Sauce Labs Backpack",
            "Sauce Labs Bike Light",
            "Sauce Labs Bolt T-Shirt",
        ]
    )

    cart_page.remove_item("sauce-labs-bike-light")

    expect(cart_page.cart_items).to_have_count(2)
    expect(cart_page.item_names).to_have_text(
        [
            "Sauce Labs Backpack",
            "Sauce Labs Bolt T-Shirt",
        ]
    )
